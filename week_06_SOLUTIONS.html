#4. What code could you add to a ucbadmit |> count(dept, gender, admit) pipeline to produce the following table. Note, this may require several steps.
  
# option 1
ucbadmit |> 
  count(dept, gender, admit) |> 
  group_by(dept, gender) |> 
  mutate(n_applied = sum(n)) |> 
  mutate(prop_admit = n/sum(n)) |> 
  filter(admit == "Admitted")

# option 2
ucbadmit |> 
  count(dept, gender, admit) |> 
  pivot_wider(names_from = admit,
              values_from = n) |> 
  mutate(n_applied = Rejected + Admitted,
         prop_admit = Admitted/n_applied) |> 
  select(-Rejected) |> 
  rename(n_admittd = Admitted)
